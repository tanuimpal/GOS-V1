-- GOS V1 â€” LOCAL-ONLY RESTRICTED BILLING READ CANDIDATE
-- NOT MIGRATION 0004. NOT APPROVED FOR PRODUCTION.
-- Dependencies: existing migrations 0001 and 0002.
-- Execute only after separate SQL review and isolated DB approval.

BEGIN;

-- Fail closed if either security role already exists.
DO $roles$
BEGIN
  IF EXISTS (
    SELECT 1 FROM pg_catalog.pg_roles
    WHERE rolname IN ('gos_billing_read_owner', 'gos_web_billing_read')
  ) THEN
    RAISE EXCEPTION 'Restricted billing-read role collision';
  END IF;
END
$roles$;

CREATE ROLE gos_billing_read_owner
  NOLOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE
  NOINHERIT NOREPLICATION NOBYPASSRLS;

CREATE ROLE gos_web_billing_read
  LOGIN NOSUPERUSER NOCREATEDB NOCREATEROLE
  NOINHERIT NOREPLICATION NOBYPASSRLS;

-- No credentials are embedded or provisioned in this candidate.

CREATE SCHEMA billing_api AUTHORIZATION CURRENT_USER;
REVOKE ALL ON SCHEMA billing_api FROM PUBLIC;

GRANT USAGE ON SCHEMA billing_api TO gos_web_billing_read;

-- Function owner receives no billing write permissions.
GRANT USAGE ON SCHEMA public, gos_billing
  TO gos_billing_read_owner;

GRANT SELECT (
  id, user_id, session_handle_hash,
  active_organization_id, expires_at, revoked_at
) ON public.auth_sessions TO gos_billing_read_owner;

GRANT SELECT (id, status)
  ON public.users TO gos_billing_read_owner;

GRANT SELECT (id, name, status)
  ON public.organizations TO gos_billing_read_owner;

GRANT SELECT (
  organization_id, user_id, role, status, accepted_at
) ON public.memberships TO gos_billing_read_owner;

GRANT SELECT (
  id, organization_id, mode, price_id, status,
  current_start, current_end, charge_at, paid_until,
  refund_blocked_until, cancel_at_period_end,
  cancel_state, create_state, last_synced_at, created_at
) ON gos_billing.subscriptions TO gos_billing_read_owner;

GRANT SELECT (id, plan_id, mode)
  ON gos_billing.prices TO gos_billing_read_owner;

GRANT SELECT (id, name)
  ON gos_billing.plans TO gos_billing_read_owner;

-- Existing billing tables use FORCE ROW LEVEL SECURITY.
-- These SELECT-only policies apply solely to the
-- dedicated, non-login SECURITY DEFINER function owner.
--
-- They do not grant access to gos_web_billing_read.
-- The function must establish session/tenant authority
-- in its own query, independently of custom GUCs.

CREATE POLICY billing_api_owner_read
  ON gos_billing.subscriptions
  FOR SELECT
  TO gos_billing_read_owner
  USING (true);

CREATE POLICY billing_api_owner_read
  ON gos_billing.prices
  FOR SELECT
  TO gos_billing_read_owner
  USING (true);

CREATE POLICY billing_api_owner_read
  ON gos_billing.plans
  FOR SELECT
  TO gos_billing_read_owner
  USING (true);

CREATE FUNCTION billing_api.read_current_subscription(
  handle_hash text,
  requested_mode text
)
RETURNS TABLE (
  id uuid,
  organization_id uuid,
  organization_name text,
  plan_name text,
  price_id uuid,
  mode text,
  status text,
  current_start timestamptz,
  current_end timestamptz,
  charge_at timestamptz,
  paid_until timestamptz,
  refund_blocked_until timestamptz,
  cancel_at_period_end boolean,
  cancel_state text,
  create_state text,
  last_synced_at timestamptz
)
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = pg_catalog
AS $function$
  SELECT
    sub.id,
    org.id,
    org.name,
    plan.name,
    sub.price_id,
    sub.mode,
    sub.status,
    sub.current_start,
    sub.current_end,
    sub.charge_at,
    sub.paid_until,
    sub.refund_blocked_until,
    sub.cancel_at_period_end,
    sub.cancel_state,
    sub.create_state,
    sub.last_synced_at
  FROM public.auth_sessions AS sess
  JOIN public.users AS usr
    ON usr.id = sess.user_id
   AND usr.status = 'ACTIVE'
  JOIN public.organizations AS org
    ON org.id = sess.active_organization_id
   AND org.status = 'ACTIVE'
  JOIN public.memberships AS membership
    ON membership.organization_id = org.id
   AND membership.user_id = usr.id
   AND membership.status = 'ACTIVE'
   AND membership.accepted_at IS NOT NULL
   AND membership.role IN ('CLIENT_OWNER', 'CLIENT_ADMIN')
  JOIN gos_billing.subscriptions AS sub
    ON sub.organization_id = org.id
   AND sub.mode = read_current_subscription.requested_mode
  JOIN gos_billing.prices AS price
    ON price.id = sub.price_id
   AND price.mode = sub.mode
  JOIN gos_billing.plans AS plan
    ON plan.id = price.plan_id
  WHERE read_current_subscription.handle_hash ~ '^[0-9a-f]{64}$'
    AND read_current_subscription.requested_mode IN ('test', 'live')
    AND sess.session_handle_hash = read_current_subscription.handle_hash
    AND sess.revoked_at IS NULL
    AND sess.expires_at > statement_timestamp()
  ORDER BY sub.created_at DESC, sub.id DESC
  LIMIT 1;
$function$;

ALTER FUNCTION billing_api.read_current_subscription(text, text)
  OWNER TO gos_billing_read_owner;

REVOKE ALL ON FUNCTION
  billing_api.read_current_subscription(text, text)
  FROM PUBLIC;

GRANT EXECUTE ON FUNCTION
  billing_api.read_current_subscription(text, text)
  TO gos_web_billing_read;

-- Deliberately no direct table grants to the web login.
-- Deliberately no general-purpose SQL API.
-- Deliberately no worker role membership.
-- Deliberately no scope-setting authorization.
-- Deliberately no modification of prior migrations.

COMMIT;

/** Only safe billing projections cross the browser boundary. Money is minor-unit text. */
export type BillingMode = 'test' | 'live';
export type SubscriptionStatus =
  | 'creating'
  | 'created'
  | 'authenticated'
  | 'active'
  | 'pending'
  | 'halted'
  | 'paused'
  | 'cancelled'
  | 'completed'
  | 'expired';

  id: string;
  code: string;
  name: string;

}

  price_id: string;
