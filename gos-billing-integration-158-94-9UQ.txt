 42: 
 43: function createScopedDatabase(
 44:   pool: Pool,
 45:   domain: 'billing' | 'social',
 46:   worker: boolean,
 47: ): BillingDatabase {
 48:   return {
 49:     async transaction(scope, work) {
 50:       if (scope.kind === 'system' && !worker)
 51:         throw new AppError(
 52:           'DATABASE_SYSTEM_SCOPE_UNAVAILABLE',
 53:           503,
 54:           'System database operations require the authenticated worker database adapter.',
 55:         );
 56:       if (worker && scope.kind !== 'system')
 57:         throw new AppError(
 58:           'DATABASE_WORKER_SCOPE_INVALID',
 59:           403,
 60:           'The worker database adapter only accepts system operations.',
 61:         );
 62:       const client = await pool.connect().catch((error) => {
 63:         throw new DatabaseError({ cause: error });
 64:       });
 65:       try {
 66:         await client.query('BEGIN');
 67:         const { rows } = await client.query<{ safe: boolean }>(
 68:           `SELECT
 69:           current_user = session_user
 70:           AND NOT EXISTS (SELECT 1 FROM pg_roles r
 71:             WHERE r.rolname IN (current_user,session_user)
 72:               AND (r.rolsuper OR r.rolbypassrls OR r.rolcreaterole OR r.rolcreatedb OR r.rolreplication))
 73:           AND NOT EXISTS (SELECT 1 FROM pg_roles granted
 74:             WHERE pg_has_role(session_user,granted.oid,'MEMBER')
 75:               AND (granted.rolsuper OR granted.rolbypassrls OR granted.rolcreaterole
 76:                 OR granted.rolcreatedb OR granted.rolreplication))
 77:           AND NOT EXISTS (SELECT 1 FROM pg_namespace n WHERE n.nspname=$1
 78:             AND pg_has_role(session_user,n.nspowner,'MEMBER'))
 79:           AND NOT EXISTS (SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
 80:             WHERE n.nspname=$1 AND pg_has_role(session_user,c.relowner,'MEMBER')) AS safe`,
 81:           ['gos_' + domain],
 82:         );
 83:         if (rows[0]?.safe !== true)
 84:           throw new AppError(
 85:             domain === 'billing' ? 'BILLING_DATABASE_ROLE_INVALID' : 'SOCIAL_DATABASE_ROLE_INVALID',
 86:             503,
 87:             'Database access is not configured safely.',
 88:           );
 89:         if (scope.kind === 'system') {
 90:           const expectedWorker = domain === 'billing' ? 'gos_billing_worker' : 'gos_social_worker';
 91:           const identity = await client.query<{ authenticated: boolean }>(
 92:             'SELECT current_user = $1 AS authenticated',
 93:             [expectedWorker],
 94:           );
 95:           if (identity.rows[0]?.authenticated !== true)
 96:             throw new AppError(
 97:               'DATABASE_WORKER_IDENTITY_INVALID',
 98:               503,
 99:               'System database operations require the authenticated worker database identity.',
100:             );
101:         }
102:         await client.query(
103:           "SELECT set_config($4,$1,true),set_config('gos.actor_id',$2,true),set_config('gos.organization_id',$3,true)",
104:           [scope.kind, scope.userId ?? '', scope.organizationId ?? '', 'gos.' + domain + '_scope'],
105:         );
106:         const result = await work(client);
107:         await client.query('COMMIT');
108:         return result;
109:       } catch (error) {
110:         await client.query('ROLLBACK').catch(() => {});
111:         if (error instanceof AppError) throw error;
112:         throw new DatabaseError({ cause: error });
113:       } finally {
114:         client.release();
115:       }
116:     },
117:     close: () => pool.end(),
118:   };
119: }
